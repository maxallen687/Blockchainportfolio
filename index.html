04 / Craft

Code your auditors will thank you for.

One feature, three layers. Explore a production-style Merkle airdrop (contract, fuzz tests and React hook) and the reasoning behind the lines that matter. It compiles, and every test passes.

// SPDX-License-Identifier: MIT
pragma solidity 0.8.28;

import {IERC20} from "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import {SafeERC20} from "@openzeppelin/contracts/token/ERC20/utils/SafeERC20.sol";
import {MerkleProof} from "@openzeppelin/contracts/utils/cryptography/MerkleProof.sol";

/// @notice Gas-efficient airdrop: one bit of storage per claim.
contract MerkleDistributor {
    using SafeERC20 for IERC20;

    error AlreadyClaimed(uint256 index);
    error InvalidProof();
    event Claimed(uint256 indexed index, address indexed account, uint256 amount);

    IERC20 public immutable token;
    bytes32 public immutable merkleRoot;
    mapping(uint256 => uint256) private claimedBitMap; // 256 flags per slot

    constructor(IERC20 token_, bytes32 merkleRoot_) {
        token = token_;
        merkleRoot = merkleRoot_;
    }

    function isClaimed(uint256 index) public view returns (bool) {
        return (claimedBitMap[index >> 8] & (1 << (index & 0xff))) != 0;
    }

    function claim(uint256 index, uint256 amount, bytes32[] calldata proof) external {
        if (isClaimed(index)) revert AlreadyClaimed(index);

        // Double-hashed leaf rules out second-preimage attacks on the tree.
        bytes32 leaf =
            keccak256(bytes.concat(keccak256(abi.encode(index, msg.sender, amount))));
        if (!MerkleProof.verifyCalldata(proof, merkleRoot, leaf)) revert InvalidProof();

        claimedBitMap[index >> 8] |= 1 << (index & 0xff); // effects first…
        token.safeTransfer(msg.sender, amount);           // …interaction last
        emit Claimed(index, msg.sender, amount);
    }
}

pragma solidity 0.8.28;

import {Test} from "forge-std/Test.sol";
import {ERC20Mock} from "./mocks/ERC20Mock.sol";
import {MerkleDistributor} from "../src/MerkleDistributor.sol";

contract MerkleDistributorTest is Test {
    ERC20Mock internal token = new ERC20Mock();

    /// Any valid leaf pays out exactly once, for every fuzzed input.
    function testFuzz_ClaimOnce(uint256 index, address user, uint96 amount) public {
        vm.assume(user != address(0));
        assumeNotForgeAddress(user);

        // Single-leaf tree: the root *is* the leaf, so the proof is empty.
        bytes32 leaf =
            keccak256(bytes.concat(keccak256(abi.encode(index, user, amount))));
        MerkleDistributor drop = new MerkleDistributor(token, leaf);
        token.mint(address(drop), amount);

        vm.prank(user);
        drop.claim(index, amount, new bytes32[](0));
        assertEq(token.balanceOf(user), amount);
        assertTrue(drop.isClaimed(index));

        vm.prank(user);
        vm.expectRevert(
            abi.encodeWithSelector(MerkleDistributor.AlreadyClaimed.selector, index)
        );
        drop.claim(index, amount, new bytes32[](0));
    }

    function testFuzz_RejectsForgedProof(bytes32 root, uint96 amount) public {
        MerkleDistributor drop = new MerkleDistributor(token, root);
        vm.expectRevert(MerkleDistributor.InvalidProof.selector);
        drop.claim(0, amount, new bytes32[](1));
    }
}
import { BaseError, ContractFunctionRevertedError, type Address, type Hex } from "viem";
import {
  useSimulateContract,
  useWaitForTransactionReceipt,
  useWriteContract,
} from "wagmi";
import { distributorAbi } from "./abi"; // `as const` → typed end to end

type Claim = { index: bigint; amount: bigint; proof: Hex[] };

export function useClaimAirdrop(distributor: Address, claim?: Claim) {
  // 1 · Simulate first: a revert surfaces before the wallet ever opens.
  const simulation = useSimulateContract({
    address: distributor,
    abi: distributorAbi,
    functionName: "claim",
    args: claim ? [claim.index, claim.amount, claim.proof] : undefined,
    query: { enabled: Boolean(claim) },
  });

  const { writeContract, data: hash, isPending } = useWriteContract();
  const receipt = useWaitForTransactionReceipt({ hash });

  // 2 · Decode custom errors into product states — never raw hex.
  const failure = simulation.error instanceof BaseError
    ? simulation.error.walk((e) => e instanceof ContractFunctionRevertedError)
    : null;
  const reason =
    failure instanceof ContractFunctionRevertedError ? failure.data?.errorName : null;

  return {
    claim: () => simulation.data && writeContract(simulation.data.request),
    alreadyClaimed: reason === "AlreadyClaimed",
    invalidProof: reason === "InvalidProof",
    status: receipt.isSuccess ? "confirmed"
      : receipt.isLoading ? "confirming"
      : isPending ? "signing" : "idle",
  } as const;
}

  1. 01

    Bitmap claims

    256 claim flags share one storage slot. Most claims flip a bit in a slot that's already in use instead of initializing a fresh one, which saves about 17k gas each.

  2. 02

    Immutable configuration

    Token and root live in bytecode, not storage. That means no cold storage reads on the hot path, and no admin key that could swap them later.

  3. 03

    Second-preimage safe

    Leaves are double-hashed (OpenZeppelin's standard tree format), so an internal node can never be replayed as a valid leaf.

  4. 04

    Custom errors

    Cheaper than revert strings and ABI-decodable, so the front-end can say “Already claimed” instead of “execution reverted”.

  5. 05

    Checks → effects → interactions

    Validate, record the claim, then transfer. Even a token with transfer hooks can't re-enter for a second payout.

  1. 01

    Property-based fuzzing

    10,000 random index / user / amount combinations per run. The property has to hold for every one of them, not just a happy path.

  2. 02

    Fixture-free trees

    A one-leaf tree's root is the leaf itself, so every fuzz run builds its own valid airdrop. No brittle JSON fixtures.

  3. 03

    Outcomes, not internals

    Asserts what users care about: the tokens arrived and the claim is recorded.

  4. 04

    Exact revert matching

    Checks the custom error and its argument. A test that passes on any revert proves nothing.

  5. 05

    Adversarial inputs

    Random roots with forged proofs must fail closed, every single time.

  1. 01

    Simulate before signing

    Every claim is dry-run against live chain state first. Users never sign (or pay gas for) a transaction that would revert.

  2. 02

    Typed end to end

    The ABI is a const, so function names and argument types are checked at compile time against the deployed contract.

  3. 03

    Readable failures

    Custom errors decode into product states like “Already claimed”, never a raw hex blob.

  4. 04

    Explicit lifecycle

    Signing → confirming → confirmed. The UI always knows exactly where the transaction is.
